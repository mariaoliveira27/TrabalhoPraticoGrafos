class Grafo {
  - vertices: Map<String, Vertice>
  - adjacencias: Map<String, List<Aresta>>
  --
  + adicionarVertice(vertice: Vertice): void
  + adicionarAresta(aresta: Aresta): void
  + getVertice(id: String): Vertice?
  + getAdjacencias(id: String): List<Aresta>
}

class Aresta {
  - id: String
  - origem: Vertice
  - destino: Vertice
  - distanciaMetros: Double
  - velocidadeKmH: Double
  - tempoSegundos: Double
  - custoReais: Double
  --
  + getPeso(criterio: String): Double
}

note right of Aresta::getPeso
  Critérios de otimização:
  * "distancia": peso = distanciaMetros
  * "tempo": peso = tempoSegundos
  * "custo": peso = custoReais (combustível)
end note

class Vertice {
  - id: String
  - nome: String?
  - latitude: Double
  - longitude: Double
  --
  + getId(): String
  + getNome(): String?
  + getLatitude(): Double
  + getLongitude(): Double
}

class ResultadoRota {
  - situacao: String
  - caminho: List<Aresta>
  - distanciaTotalMetros: Double?
  - tempoTotalSegundos: Double?
  - custoTotalReais: Double?
  - nosExpandidos: Integer
  --
  + getDistanciaKm(): Double?
  + getTempoMinutos(): Double?
}
